115/// The `proxy_pool` parameter of the Web Scraping API, the Crawler and the
116/// Screenshot API: [`ProxyPool::Basic`], [`ProxyPool::Unblocker`],
117/// [`ProxyPool::UnblockerUltra`] or [`ProxyPool::Auto`], each of which sets
118/// both the proxies and the Unblocker, or [`ProxyPool::PublicTorPool`], the
119/// Tor network. The Screenshot API takes the first four and ignores any other
120/// value. Cloud Browser has pools of its own (see
121/// [`BrowserConfig`](crate::BrowserConfig)).
122///
123/// The pool names written before the proxy pools
124/// ([`ProxyPool::PublicDatacenterPool`], [`ProxyPool::PublicResidentialPool`])
125/// are still accepted. The migration guide covers them:
126/// <https://scrapfly.io/new-pricing#legacy-spellings>
127///
128/// Non-exhaustive: the API grows this list, and a new member must not break
129/// a downstream `match`.
130#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
131#[serde(rename_all = "snake_case")]
132#[non_exhaustive]
133pub enum ProxyPool {
134    /// `basic`: datacenter proxies, Unblocker off.
135    Basic,
136    /// `unblocker`: residential proxies with the Unblocker (anti-bot bypass)
137    /// on.
138    Unblocker,
139    /// `unblocker_ultra`: residential proxies with the Unblocker on. The only
140    /// pool accepted by the targets and anti-bot protections that have a
141    /// price of their own: those refuse `basic` and `unblocker` with
142    /// `ERR::PROXY::UNBLOCKER_ULTRA_REQUIRED` (HTTP 422, not billed).
143    UnblockerUltra,
144    /// `auto`: the API picks the pool before the request, from the target's
145    /// profile: the Unblocker on datacenter proxies, or the pool the target
146    /// needs. Accepted on every target, the ones that require
147    /// [`ProxyPool::UnblockerUltra`] included, where their price is added;
148    /// priced as the pool it runs on.
149    Auto,
150    /// Former name of the datacenter pool.
151    PublicDatacenterPool,
152    /// Former name of the residential pool.
153    PublicResidentialPool,
154    /// `public_tor_pool`: egress through the Tor network, the only pool that
155    /// reaches .onion hidden services; also used for OSINT.
156    PublicTorPool,
157}
158
159impl ProxyPool {
160    /// Wire-format string.
161    pub fn as_str(&self) -> &'static str {
162        match self {
163            Self::Basic => "basic",
164            Self::Unblocker => "unblocker",
165            Self::UnblockerUltra => "unblocker_ultra",
166            Self::Auto => "auto",
167            Self::PublicDatacenterPool => "public_datacenter_pool",
168            Self::PublicResidentialPool => "public_residential_pool",
169            Self::PublicTorPool => "public_tor_pool",
170        }
171    }
172
173    /// Whether this is one of the pool names written before the proxy pools
174    /// ([`ProxyPool::PublicDatacenterPool`], [`ProxyPool::PublicResidentialPool`]).
175    /// [`ProxyPool::PublicTorPool`] is not one: the Tor pool keeps its name.
176    pub fn is_legacy(&self) -> bool {
177        matches!(
178            self,
179            Self::PublicDatacenterPool | Self::PublicResidentialPool
180        )
181    }
182}
183
184/// Lets the crawler builder, which takes the pool as a string, accept a `ProxyPool`.
185impl From<ProxyPool> for String {
186    fn from(p: ProxyPool) -> Self {
187        p.as_str().to_owned()
188    }
189}

462/// External secret manager a Cloud Browser vault mirrors.
463///
464/// The server keys its decoder on this discriminator and rejects anything
465/// outside the registry (`pkg/vault` `ErrUnknownLinkedService`), so the set is
466/// closed to what `linkedServiceRegistry` holds.
467///
468/// Closed on purpose, unlike the Python, Go and TypeScript SDKs which also take
469/// a raw string: a provider registered server-side needs a release of this crate
470/// before a caller here can name it.
471#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
472pub enum VaultLinkedService {
473    /// 1Password, via a service-account token.
474    #[serde(rename = "1password")]
475    OnePassword,
476}
477
478impl VaultLinkedService {
479    /// Wire-format string.
480    pub fn as_str(&self) -> &'static str {
481        match self {
482            Self::OnePassword => "1password",
483        }
484    }
