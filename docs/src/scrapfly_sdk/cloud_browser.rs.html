140/// Return the deterministic project salt for an api key (`sha256(api_key)[:8]`).
141/// Matches the `X-Browser-Project-Salt` response header returned on
142/// VNC-enabled Cloud Browser upgrades, where the salt is also the VNC
143/// password prefix (`<salt>-<password>`).
144pub fn project_salt(api_key: &str) -> String {
145    use sha2::{Digest, Sha256};
146    let digest = Sha256::digest(api_key.as_bytes());
147    hex::encode(&digest[..4])
148}

150impl BrowserConfig {
151    /// Return the password a native VNC client must type to attach to a
152    /// session created with this config: `<project_salt>-<vnc_password>`.
153    ///
154    /// Required by the VNC TCP endpoint (port 5901), which the server salts at
155    /// allocation. The WebSocket endpoint `/run/<run_id>/vnc` takes the raw
156    /// `vnc_password` instead. `None` when the config carries no VNC credential
157    /// to salt.
158    pub fn vnc_client_password(&self, api_key: &str) -> Option<String> {
159        if !self.enable_vnc {
160            return None;
161        }
162        self.vnc_password
163            .as_deref()
164            .filter(|pw| !pw.is_empty())
165            .map(|pw| format!("{}-{}", project_salt(api_key), pw))
166    }
167}

173/// Normalize an arbitrary Cloud Browser host to a `wss://` URL, regardless of
174/// the scheme the caller configured. Accepted input schemes: `https://`
175/// (default), `wss://`, `ws://`, `http://`, and bare `host[:port]`. Mirrors
176/// `sdk/go/cloud_browser.go::wsBase`.
177fn ws_base(host: &str) -> String {
178    if let Some(rest) = host.strip_prefix("wss://") {
179        format!("wss://{}", rest)
180    } else if let Some(rest) = host.strip_prefix("ws://") {
181        format!("ws://{}", rest)
182    } else if let Some(rest) = host.strip_prefix("https://") {
183        format!("wss://{}", rest)
184    } else if let Some(rest) = host.strip_prefix("http://") {
185        format!("ws://{}", rest)
186    } else {
187        format!("wss://{}", host)
188    }
189}
190
191/// Normalize an arbitrary Cloud Browser host to its REST form (`https://` or
192/// `http://`). Callers typically configure a `wss://` / `ws://` host (the CDP
193/// entry point); the REST endpoints (`/unblock`, `/session/.../stop`) live on
194/// the HTTP-equivalent origin. Mirrors `sdk/go/cloud_browser.go::restBase`.
195fn rest_base(host: &str) -> String {
196    if let Some(rest) = host.strip_prefix("wss://") {
197        format!("https://{}", rest)
198    } else if let Some(rest) = host.strip_prefix("ws://") {
199        format!("http://{}", rest)
200    } else if host.starts_with("https://") || host.starts_with("http://") {
201        host.to_string()
202    } else {
203        format!("https://{}", host)
204    }
205}

271        }
272        if let Some(v) = &config.country {
273            pairs.push(("country".into(), v.clone()));
274        }
275        if let Some(v) = &config.lang {
276            pairs.push(("lang".into(), v.clone()));
277        }
278        if !config.languages.is_empty() {
279            pairs.push(("languages".into(), config.languages.join(",")));
280        }
281        if !config.extensions.is_empty() {
282            pairs.push(("extensions".into(), config.extensions.join(",")));
283        }
284        if let Some(v) = &config.session {
285            pairs.push(("session".into(), v.clone()));
286        }
287        if let Some(v) = config.timeout {
288            pairs.push(("timeout".into(), v.to_string()));
289        }
290        if config.block_images {
291            pairs.push(("block_images".into(), "true".into()));
292        }
293        if config.block_styles {
294            pairs.push(("block_styles".into(), "true".into()));
295        }
296        if config.block_fonts {
297            pairs.push(("block_fonts".into(), "true".into()));
298        }
299        if config.block_media {
300            pairs.push(("block_media".into(), "true".into()));
301        }
302        if config.screenshot {
303            pairs.push(("screenshot".into(), "true".into()));
304        }
305        if config.cache {
306            pairs.push(("cache".into(), "true".into()));
307        }
308        if config.blacklist {
309            pairs.push(("blacklist".into(), "true".into()));
310        }
311        if config.debug {
312            pairs.push(("debug".into(), "true".into()));
313        }
314        if let Some(v) = &config.target_url {
315            pairs.push(("target_url".into(), v.clone()));

325        }
326        if config.enable_mcp {
327            pairs.push(("enable_mcp".into(), "true".into()));

329        if config.solve_captcha {
330            pairs.push(("solve_captcha".into(), "true".into()));

332        if let Some(v) = &config.vault {

334        }
335        if let Some(v) = &config.vault_key {
336            pairs.push(("vault_key".into(), v.clone()));
337        }
338        if config.enable_vnc {
339            pairs.push(("enable_vnc".into(), "true".into()));

341        if let Some(v) = &config.vnc_password {

345            pairs.push(("enable_rtc".into(), "true".into()));

379    /// Call `POST /unblock` to bypass anti-bot protection.
380    pub async fn cloud_browser_unblock(
381        &self,
382        config: &UnblockConfig,
383    ) -> Result<UnblockResult, ScrapflyError> {
384        let url = format!(
385            "{}/unblock?key={}",
386            rest_base(self.cloud_browser_host()),
387            self.api_key()
388        );
389        let url = Url::parse(&url)
390            .map_err(|e| ScrapflyError::Config(format!("invalid unblock url: {}", e)))?;
391        let body = serde_json::to_vec(config)?;
392        let mut headers = HeaderMap::new();
393        headers.insert(CONTENT_TYPE, HeaderValue::from_static("application/json"));
394        let resp = self
395            .send_with_retry(Method::POST, url, Some(headers), Some(body))
396            .await?;
397        let status = resp.status().as_u16();
398        let body = resp.bytes().await.map_err(ScrapflyError::Transport)?;
399        if status != 200 {
400            return Err(from_response(status, &body, 0, false));
401        }
402        Ok(serde_json::from_slice(&body)?)
403    }

445    /// Upload a browser extension from a local file (.zip or .crx).
446    pub async fn cloud_browser_extension_upload(
447        &self,
448        file_path: &std::path::Path,
449    ) -> Result<serde_json::Value, ScrapflyError> {
450        let url = format!(
451            "{}/extension?key={}",
452            rest_base(self.cloud_browser_host()),
453            self.api_key()
454        );
455        let url = Url::parse(&url)
456            .map_err(|e| ScrapflyError::Config(format!("invalid extension url: {}", e)))?;
457        let file_bytes = std::fs::read(file_path)
458            .map_err(|e| ScrapflyError::Config(format!("failed to read extension file: {}", e)))?;
459        let file_name = file_path
460            .file_name()
461            .and_then(|n| n.to_str())
462            .unwrap_or("extension.zip")
463            .to_string();
464        // Build multipart body manually (reqwest multipart feature not enabled)
465        let boundary = format!(
466            "----ScrapflyBoundary{}",
467            std::time::SystemTime::now()
468                .duration_since(std::time::UNIX_EPOCH)
469                .unwrap_or_default()
470                .as_millis()
471        );
472        let mut body = Vec::new();
473        body.extend_from_slice(format!("--{}\r\n", boundary).as_bytes());
474        body.extend_from_slice(
475            format!(
476                "Content-Disposition: form-data; name=\"file\"; filename=\"{}\"\r\n\
477                 Content-Type: application/octet-stream\r\n\r\n",
478                file_name
479            )
480            .as_bytes(),
481        );
482        body.extend_from_slice(&file_bytes);
483        body.extend_from_slice(format!("\r\n--{}--\r\n", boundary).as_bytes());
484        let mut headers = HeaderMap::new();
485        headers.insert(
486            CONTENT_TYPE,
487            HeaderValue::from_str(&format!("multipart/form-data; boundary={}", boundary))
488                .map_err(|e| ScrapflyError::Config(format!("invalid content-type: {}", e)))?,
489        );
490        let resp = self
491            .send_with_retry(Method::POST, url, Some(headers), Some(body))
492            .await?;
493        let status = resp.status().as_u16();
494        let body = resp.bytes().await.map_err(ScrapflyError::Transport)?;
495        if status != 200 && status != 201 {
496            return Err(from_response(status, &body, 0, false));
497        }
498        Ok(serde_json::from_slice(&body)?)
499    }

537    /// Create a new credential vault. The server returns a freshly
538    /// generated vault key under the `key` field of the response — this
539    /// is the only time the key is shown. Save it locally; the server
540    /// cannot recover it.
541    pub async fn cloud_browser_vault_create(
542        &self,
543        name: &str,
544        description: Option<&str>,
545    ) -> Result<serde_json::Value, ScrapflyError> {
546        let url = format!(
547            "{}/vault?key={}",
548            rest_base(self.cloud_browser_host()),
549            self.api_key()

874    /// Poll the playback endpoint until the recording resolves to a terminal
875    /// state (`ready` or `unavailable`) or the timeout elapses. Honours the
876    /// server's `retry_after_ms` hint when present.
877    pub async fn cloud_browser_wait_for_playback(
878        &self,
879        run_id: &str,
880        timeout: std::time::Duration,
881        fallback_interval: std::time::Duration,
882    ) -> Result<serde_json::Value, ScrapflyError> {
883        let deadline = std::time::Instant::now() + timeout;
884        loop {
885            let playback = self.cloud_browser_playback(run_id).await?;
886            let status = playback
887                .get("status")
888                .and_then(|v| v.as_str())
889                .unwrap_or("");
890            if status != "uploading" {
891                return Ok(playback);
892            }
893            let remaining = deadline.saturating_duration_since(std::time::Instant::now());
894            if remaining.is_zero() {
895                return Ok(playback);
896            }
897            let retry_after_ms = playback
898                .get("retry_after_ms")
899                .and_then(|v| v.as_u64())
900                .unwrap_or(0);
901            let next = if retry_after_ms > 0 {
902                std::time::Duration::from_millis(retry_after_ms)
903            } else {
904                fallback_interval
905            };
906            tokio::time::sleep(next.min(remaining)).await;
907        }
908    }
