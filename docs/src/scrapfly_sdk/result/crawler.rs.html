97impl CrawlerStatus {
98    /// True while still pending or running.
99    pub fn is_running(&self) -> bool {
100        self.status == status::PENDING || self.status == status::RUNNING
101    }
102    /// True when terminated successfully.
103    pub fn is_complete(&self) -> bool {
104        self.status == status::DONE && self.is_success == Some(true)
105    }
106    /// True when terminated with failure.
107    pub fn is_failed(&self) -> bool {
108        self.status == status::DONE && self.is_success == Some(false)
109    }
110    /// True when cancelled by the user.
111    pub fn is_cancelled(&self) -> bool {
112        self.status == status::CANCELLED
113    }

173impl CrawlerRefreshEntry {
174    /// Pages this run actually touched. Zero means the site stood still and

181/// Coerce JSON `null | absent | array` into a plain `Vec<T>`, where null and
182/// absent both collapse to empty. Every list on this surface is rendered from
183/// a Go slice declared without `omitempty`, so a nil slice reaches the wire as
184/// `null`; `serde(default)` alone covers only the absent key and would reject
185/// the whole envelope. Mirrors Go's nil-slice-is-an-empty-slice semantics.
186fn null_as_empty_vec<'de, D, T>(deserializer: D) -> Result<Vec<T>, D::Error>
187where
188    D: serde::Deserializer<'de>,
189    T: Deserialize<'de>,
190{
191    Ok(Option::<Vec<T>>::deserialize(deserializer)?.unwrap_or_default())
192}

194/// The `refresh` block of a crawl, carried by `GET /crawl/{uuid}/status` and
195/// returned by the three refresh calls.
196#[derive(Debug, Clone, Deserialize, Default)]
197pub struct CrawlerRefreshState {
198    /// Whether the crawl re-scrapes itself on a period.
199    #[serde(default)]
200    pub enabled: bool,
201    /// Period between runs, 0 when disabled.
202    #[serde(default)]
203    pub interval_seconds: u32,
204    /// One of [`refresh_status`].
205    #[serde(default)]
206    pub status: String,
207    /// Refresh runs completed so far.
208    #[serde(default)]
209    pub generation: u32,
210    /// ISO-8601 timestamp of the last completed run.
211    #[serde(default)]
212    pub last_run_at: Option<String>,
213    /// ISO-8601 timestamp of the next due run, `None` when disabled.
214    #[serde(default)]
215    pub next_run_at: Option<String>,
216    /// ISO-8601 start of the run in flight, `None` unless `status` is
217    /// `RUNNING`. Only `GET /crawl/{uuid}/status` reports it: the three
218    /// refresh calls render a typed block that omits the key, where it reads
219    /// `None` whatever the schedule is doing.
220    #[serde(default)]
221    pub started_at: Option<String>,
222    /// Failure reason when `status` is `FAILED`.
223    #[serde(default)]
224    pub error: Option<String>,
225    /// Failed runs since the last success. Route-scoped like `started_at`, so
226    /// a zero read off a refresh call means "not reported", not "no failures".
227    #[serde(default)]
228    pub consecutive_failures: u32,
229    /// Newest last, capped at the 50 most recent runs.
230    #[serde(default, deserialize_with = "null_as_empty_vec")]
231    pub history: Vec<CrawlerRefreshEntry>,
232}

234impl CrawlerRefreshState {
235    /// Decode a refresh envelope.
236    ///
237    /// The three refresh calls answer with the state at the top level;
238    /// `GET /crawl/{uuid}/status` nests it under `refresh`. Accepting both
239    /// means the SDK never has to guess which call produced the bytes.
240    pub fn from_envelope(bytes: &[u8]) -> Result<Self, serde_json::Error> {
241        let mut envelope: serde_json::Value = serde_json::from_slice(bytes)?;
242        let nested = envelope
243            .get_mut("refresh")
244            .filter(|block| block.is_object())
245            .map(serde_json::Value::take);
246        serde_json::from_value(nested.unwrap_or(envelope))
247    }
248
249    /// True while a refresh run is in flight.
250    pub fn is_running(&self) -> bool {
251        self.status == refresh_status::RUNNING
252    }
253
254    /// Most recent timeline row, `None` before the first run.
255    pub fn last_run(&self) -> Option<&CrawlerRefreshEntry> {
256        self.history.last()
257    }
258}

593/// Streaming response from `GET /crawl/{uuid}/urls`.
594#[derive(Debug, Clone, Default)]
595pub struct CrawlerUrls {
596    /// URL entries on this page.
597    pub urls: Vec<CrawlerUrlEntry>,
598    /// Page number.
599    pub page: u32,
600    /// Page size.
601    pub per_page: u32,
602}
603
604impl CrawlerUrls {
605    /// Parse a `text/plain` body into a [`CrawlerUrls`]. Mirrors
606    /// `sdk/go/result_crawler.go::parseCrawlerURLs`.
607    pub fn from_text(body: &str, status_hint: &str, page: u32, per_page: u32) -> Self {
608        let mut urls = Vec::new();
609        for raw_line in body.split('\n') {
610            let line = raw_line.trim();
611            if line.is_empty() {
612                continue;
613            }
614            if status_hint == "visited" || status_hint == "pending" {
615                urls.push(CrawlerUrlEntry {
616                    url: line.to_string(),
617                    status: status_hint.to_string(),
618                    reason: String::new(),
619                });
620                continue;
621            }
622            if let Some(idx) = line.find(',') {
623                urls.push(CrawlerUrlEntry {
624                    url: line[..idx].to_string(),
625                    status: status_hint.to_string(),
626                    reason: line[idx + 1..].to_string(),
627                });
628            } else {
629                urls.push(CrawlerUrlEntry {
630                    url: line.to_string(),
631                    status: status_hint.to_string(),
632                    reason: String::new(),
633                });
634            }
635        }
636        Self {
637            urls,
638            page,
639            per_page,
640        }
641    }
642}

649    /// that no template matched); the SDK flattens `null → ""` so consumers
650    /// always get a string and can check emptiness. Mirrors Go's map[string]string
651    /// zero-value semantics.
652    #[serde(default, deserialize_with = "deserialize_contents_map")]
653    pub contents: BTreeMap<String, BTreeMap<String, String>>,
654    /// Pagination links.
655    #[serde(default)]
656    pub links: CrawlerContentsLinks,
657}
658
659/// Deserialize `{url: {format: string|null}}` tolerating `null` inner values
660/// by mapping them to the empty string.
661fn deserialize_contents_map<'de, D>(
662    deserializer: D,
663) -> Result<BTreeMap<String, BTreeMap<String, String>>, D::Error>
664where
665    D: serde::Deserializer<'de>,
666{
667    let raw: BTreeMap<String, BTreeMap<String, Option<String>>> =
668        BTreeMap::deserialize(deserializer)?;
669    Ok(raw
670        .into_iter()
671        .map(|(url, by_format)| {
672            (
673                url,
674                by_format
675                    .into_iter()
676                    .map(|(fmt, body)| (fmt, body.unwrap_or_default()))
677                    .collect(),
678            )
679        })
680        .collect())
681}

830/// Deserialize `{key: string|null}` into a plain map, collapsing a null map
831/// and a null value alike to their empty form. Same zero-value contract as
832/// [`null_as_empty_string`], one level up.
833fn null_as_empty_string_map<'de, D>(deserializer: D) -> Result<BTreeMap<String, String>, D::Error>
834where
835    D: serde::Deserializer<'de>,

837    let raw: Option<BTreeMap<String, Option<String>>> = Option::deserialize(deserializer)?;

839        .unwrap_or_default()
840        .into_iter()
841        .map(|(key, value)| (key, value.unwrap_or_default()))
842        .collect())
843}

985/// One decoded crawler webhook delivery.
986///
987/// The envelope is always `{"event": ..., "payload": {...}}` and the event
988/// name is what decides the payload shape, so the enum is tagged on it: an
989/// event this SDK does not know rejects rather than decoding into the wrong
990/// variant.
991#[derive(Debug, Clone, Deserialize)]
992#[serde(tag = "event", content = "payload", rename_all = "snake_case")]
993pub enum CrawlerWebhook {
994    /// `crawler_started`.
995    CrawlerStarted(CrawlerLifecyclePayload),
996    /// `crawler_url_visited`.
997    CrawlerUrlVisited(CrawlerUrlVisitedPayload),
998    /// `crawler_url_skipped`.
999    CrawlerUrlSkipped(CrawlerUrlSkippedPayload),
1000    /// `crawler_url_discovered`.
1001    CrawlerUrlDiscovered(CrawlerUrlDiscoveredPayload),
1002    /// `crawler_url_failed`.
1003    CrawlerUrlFailed(CrawlerUrlFailedPayload),
1004    /// `crawler_stopped`.
1005    CrawlerStopped(CrawlerLifecyclePayload),
1006    /// `crawler_cancelled`.
1007    CrawlerCancelled(CrawlerLifecyclePayload),
1008    /// `crawler_finished`.
1009    CrawlerFinished(CrawlerLifecyclePayload),
1010    /// `crawler_search_ready`.
1011    CrawlerSearchReady(CrawlerSearchPayload),
1012    /// `crawler_search_failed`.
1013    CrawlerSearchFailed(CrawlerSearchPayload),
1014    /// `crawler_updated`.
1015    CrawlerUpdated(CrawlerUpdatedPayload),
1016}
1017
1018impl CrawlerWebhook {
1019    /// Decode a webhook request body.
1020    ///
1021    /// ```no_run
1022    /// use scrapfly_sdk::result::crawler::CrawlerWebhook;
1023    ///
1024    /// # fn handle(body: &[u8]) -> Result<(), serde_json::Error> {
1025    /// match CrawlerWebhook::from_slice(body)? {
1026    ///     CrawlerWebhook::CrawlerFinished(payload) => {
1027    ///         println!("visited {}", payload.common.state.urls_visited);
1028    ///     }
1029    ///     CrawlerWebhook::CrawlerUpdated(payload) => {
1030    ///         println!("re-indexed {}", payload.documents.updated.len());
1031    ///     }
1032    ///     other => println!("{:?}", other.event().as_str()),
1033    /// }
1034    /// # Ok(()) }
1035    /// ```
1036    pub fn from_slice(body: &[u8]) -> Result<Self, serde_json::Error> {
1037        serde_json::from_slice(body)
1038    }
1039
1040    /// The event that produced this delivery.
1041    pub fn event(&self) -> CrawlerWebhookEvent {
1042        match self {
1043            Self::CrawlerStarted(_) => CrawlerWebhookEvent::CrawlerStarted,
1044            Self::CrawlerUrlVisited(_) => CrawlerWebhookEvent::CrawlerUrlVisited,
1045            Self::CrawlerUrlSkipped(_) => CrawlerWebhookEvent::CrawlerUrlSkipped,
1046            Self::CrawlerUrlDiscovered(_) => CrawlerWebhookEvent::CrawlerUrlDiscovered,
1047            Self::CrawlerUrlFailed(_) => CrawlerWebhookEvent::CrawlerUrlFailed,
1048            Self::CrawlerStopped(_) => CrawlerWebhookEvent::CrawlerStopped,
1049            Self::CrawlerCancelled(_) => CrawlerWebhookEvent::CrawlerCancelled,
1050            Self::CrawlerFinished(_) => CrawlerWebhookEvent::CrawlerFinished,
1051            Self::CrawlerSearchReady(_) => CrawlerWebhookEvent::CrawlerSearchReady,
1052            Self::CrawlerSearchFailed(_) => CrawlerWebhookEvent::CrawlerSearchFailed,
1053            Self::CrawlerUpdated(_) => CrawlerWebhookEvent::CrawlerUpdated,
1054        }
1055    }
1056
1057    /// The fields every payload carries, whatever the event.
1058    pub fn common(&self) -> &CrawlerWebhookCommon {
1059        match self {
1060            Self::CrawlerStarted(payload)
1061            | Self::CrawlerStopped(payload)
1062            | Self::CrawlerCancelled(payload)
1063            | Self::CrawlerFinished(payload) => &payload.common,
1064            Self::CrawlerUrlVisited(payload) => &payload.common,
1065            Self::CrawlerUrlSkipped(payload) => &payload.common,
1066            Self::CrawlerUrlDiscovered(payload) => &payload.common,
1067            Self::CrawlerUrlFailed(payload) => &payload.common,
1068            Self::CrawlerSearchReady(payload) | Self::CrawlerSearchFailed(payload) => {
1069                &payload.common
1070            }
1071            Self::CrawlerUpdated(payload) => &payload.common,
1072        }
1073    }
