141/// Return the deterministic project salt for an api key (`sha256(api_key)[:8]`).
142/// Matches the `X-Browser-Project-Salt` response header returned on
143/// VNC-enabled Cloud Browser upgrades, where the salt is also the VNC
144/// password prefix (`<salt>-<password>`).
145pub fn project_salt(api_key: &str) -> String {
146    use sha2::{Digest, Sha256};
147    let digest = Sha256::digest(api_key.as_bytes());
148    hex::encode(&digest[..4])
149}

151impl BrowserConfig {
152    /// Return the password a native VNC client must type to attach to a
153    /// session created with this config: `<project_salt>-<vnc_password>`.
154    ///
155    /// Required by the VNC TCP endpoint (port 5901), which the server salts at
156    /// allocation. The WebSocket endpoint `/run/<run_id>/vnc` takes the raw
157    /// `vnc_password` instead. `None` when the config carries no VNC credential
158    /// to salt.
159    pub fn vnc_client_password(&self, api_key: &str) -> Option<String> {
160        if !self.enable_vnc {
161            return None;
162        }
163        self.vnc_password
164            .as_deref()
165            .filter(|pw| !pw.is_empty())
166            .map(|pw| format!("{}-{}", project_salt(api_key), pw))
167    }
168}

174/// Normalize an arbitrary Cloud Browser host to a `wss://` URL, regardless of
175/// the scheme the caller configured. Accepted input schemes: `https://`
176/// (default), `wss://`, `ws://`, `http://`, and bare `host[:port]`. Mirrors
177/// `sdk/go/cloud_browser.go::wsBase`.
178fn ws_base(host: &str) -> String {
179    if let Some(rest) = host.strip_prefix("wss://") {
180        format!("wss://{}", rest)
181    } else if let Some(rest) = host.strip_prefix("ws://") {
182        format!("ws://{}", rest)
183    } else if let Some(rest) = host.strip_prefix("https://") {
184        format!("wss://{}", rest)
185    } else if let Some(rest) = host.strip_prefix("http://") {
186        format!("ws://{}", rest)
187    } else {
188        format!("wss://{}", host)
189    }
190}

192/// Normalize an arbitrary Cloud Browser host to its REST form (`https://` or
193/// `http://`). Callers typically configure a `wss://` / `ws://` host (the CDP
194/// entry point); the REST endpoints (`/unblock`, `/session/.../stop`) live on
195/// the HTTP-equivalent origin. Mirrors `sdk/go/cloud_browser.go::restBase`.
196fn rest_base(host: &str) -> String {
197    if let Some(rest) = host.strip_prefix("wss://") {
198        format!("https://{}", rest)
199    } else if let Some(rest) = host.strip_prefix("ws://") {
200        format!("http://{}", rest)
201    } else if host.starts_with("https://") || host.starts_with("http://") {
202        host.to_string()
203    } else {
204        format!("https://{}", host)
205    }
206}

208/// Drop `token_item_id` from a caller-supplied `linked_service_data`.
209///
210/// The server owns that id on both /service routes: the link seals the token
211/// blob and binds the id it just created, and the update keeps the stored one,
212/// so a body can never redirect the token lookup at another row. A
213/// get-modify-update round trip carries the field back, and sending it would
214/// suggest a caller can choose it.
215fn strip_token_item_id(mut data: serde_json::Value) -> serde_json::Value {
216    if let Some(map) = data.as_object_mut() {
217        map.remove("token_item_id");
218    }
219    data
220}

290        if let Some(v) = &config.lang {

295        }

297            pairs.push(("extensions".into(), config.extensions.join(",")));

301        }
302        if let Some(v) = config.timeout {

304        }
305        if config.block_images {
306            pairs.push(("block_images".into(), "true".into()));
307        }
308        if config.block_styles {
309            pairs.push(("block_styles".into(), "true".into()));
310        }
311        if config.block_fonts {
312            pairs.push(("block_fonts".into(), "true".into()));

326        if config.debug {
327            pairs.push(("debug".into(), "true".into()));

329        if let Some(v) = &config.target_url {

338        if let Some(v) = &config.byop_proxy {

340        }

342            pairs.push(("enable_mcp".into(), "true".into()));

344        if config.solve_captcha {

394    /// Call `POST /unblock` to bypass anti-bot protection.
395    pub async fn cloud_browser_unblock(
396        &self,
397        config: &UnblockConfig,
398    ) -> Result<UnblockResult, ScrapflyError> {
399        let url = format!(
400            "{}/unblock?key={}",
401            rest_base(self.cloud_browser_host()),
402            self.api_key()
403        );
404        let url = Url::parse(&url)
405            .map_err(|e| ScrapflyError::Config(format!("invalid unblock url: {}", e)))?;
406        let body = serde_json::to_vec(config)?;
407        let mut headers = HeaderMap::new();
408        headers.insert(CONTENT_TYPE, HeaderValue::from_static("application/json"));
409        let resp = self
410            .send_with_retry(Method::POST, url, Some(headers), Some(body))
411            .await?;
412        let status = resp.status().as_u16();
413        let body = resp.bytes().await.map_err(ScrapflyError::Transport)?;
414        if status != 200 {
415            return Err(from_response(status, &body, 0, false));
416        }
417        Ok(serde_json::from_slice(&body)?)
418    }

460    /// Upload a browser extension from a local file (.zip or .crx).
461    pub async fn cloud_browser_extension_upload(
462        &self,
463        file_path: &std::path::Path,
464    ) -> Result<serde_json::Value, ScrapflyError> {
465        let url = format!(
466            "{}/extension?key={}",
467            rest_base(self.cloud_browser_host()),
468            self.api_key()
469        );
470        let url = Url::parse(&url)
471            .map_err(|e| ScrapflyError::Config(format!("invalid extension url: {}", e)))?;
472        let file_bytes = std::fs::read(file_path)
473            .map_err(|e| ScrapflyError::Config(format!("failed to read extension file: {}", e)))?;
474        let file_name = file_path
475            .file_name()
476            .and_then(|n| n.to_str())
477            .unwrap_or("extension.zip")
478            .to_string();
479        // Build multipart body manually (reqwest multipart feature not enabled)
480        let boundary = format!(
481            "----ScrapflyBoundary{}",
482            std::time::SystemTime::now()
483                .duration_since(std::time::UNIX_EPOCH)
484                .unwrap_or_default()
485                .as_millis()
486        );
487        let mut body = Vec::new();
488        body.extend_from_slice(format!("--{}\r\n", boundary).as_bytes());
489        body.extend_from_slice(
490            format!(
491                "Content-Disposition: form-data; name=\"file\"; filename=\"{}\"\r\n\
492                 Content-Type: application/octet-stream\r\n\r\n",
493                file_name
494            )
495            .as_bytes(),
496        );
497        body.extend_from_slice(&file_bytes);
498        body.extend_from_slice(format!("\r\n--{}--\r\n", boundary).as_bytes());
499        let mut headers = HeaderMap::new();
500        headers.insert(
501            CONTENT_TYPE,
502            HeaderValue::from_str(&format!("multipart/form-data; boundary={}", boundary))
503                .map_err(|e| ScrapflyError::Config(format!("invalid content-type: {}", e)))?,
504        );
505        let resp = self
506            .send_with_retry(Method::POST, url, Some(headers), Some(body))
507            .await?;
508        let status = resp.status().as_u16();
509        let body = resp.bytes().await.map_err(ScrapflyError::Transport)?;
510        if status != 200 && status != 201 {
511            return Err(from_response(status, &body, 0, false));
512        }
513        Ok(serde_json::from_slice(&body)?)
514    }

543    // The vault key returned by `cloud_browser_vault_create` /

1098        // An empty request body is what selects the sealed-token probe, so
1099        // nothing is sent when neither field is set.
1100        let body_bytes = if body_map.is_empty() {
1101            None
1102        } else {
1103            headers.insert(CONTENT_TYPE, HeaderValue::from_static("application/json"));
1104            Some(serde_json::to_vec(&serde_json::Value::Object(body_map))?)
1105        };
1106        let resp = self
1107            .send_with_retry(Method::POST, url, Some(headers), body_bytes)
1108            .await?;
1109        let status = resp.status().as_u16();
1110        let body = resp.bytes().await.map_err(ScrapflyError::Transport)?;
1111        if status != 200 {
1112            return Err(from_response(status, &body, 0, false));
1113        }
1114        Ok(serde_json::from_slice(&body)?)
1115    }

1141    /// Poll the playback endpoint until the recording resolves to a terminal
1142    /// state (`ready` or `unavailable`) or the timeout elapses. Honours the
1143    /// server's `retry_after_ms` hint when present.
1144    pub async fn cloud_browser_wait_for_playback(
1145        &self,
1146        run_id: &str,
1147        timeout: std::time::Duration,
1148        fallback_interval: std::time::Duration,
1149    ) -> Result<serde_json::Value, ScrapflyError> {
1150        let deadline = std::time::Instant::now() + timeout;
1151        loop {
1152            let playback = self.cloud_browser_playback(run_id).await?;
1153            let status = playback
1154                .get("status")
1155                .and_then(|v| v.as_str())
1156                .unwrap_or("");
1157            if status != "uploading" {
1158                return Ok(playback);
1159            }
1160            let remaining = deadline.saturating_duration_since(std::time::Instant::now());
1161            if remaining.is_zero() {
1162                return Ok(playback);
1163            }
1164            let retry_after_ms = playback
1165                .get("retry_after_ms")
1166                .and_then(|v| v.as_u64())
1167                .unwrap_or(0);
1168            let next = if retry_after_ms > 0 {
1169                std::time::Duration::from_millis(retry_after_ms)
1170            } else {
1171                fallback_interval
1172            };
1173            tokio::time::sleep(next.min(remaining)).await;
1174        }
1175    }
