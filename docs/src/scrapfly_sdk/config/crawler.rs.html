10/// Auto-refresh interval floor, in seconds (1 hour).
11///
12/// The floor decides the cost: a crawl refreshing every minute re-scrapes the
13/// whole site 1,440 times a day.
14pub const REFRESH_MIN_INTERVAL: u32 = 3600;
15/// Auto-refresh interval ceiling, in seconds (90 days).
16pub const REFRESH_MAX_INTERVAL: u32 = 90 * 24 * 3600;

18/// Configuration for a `POST /crawl` request.
19///
20/// Exactly one URL source must be provided: `url` (seed crawl with
21/// discovery), `url_list` (in-memory list, no discovery), or
22/// `remote_url_list` (URL of a hosted text file fetched at crawl start, no
23/// discovery). Other fields default to server-side values when zero.
24#[derive(Debug, Clone, Default, Serialize)]
25pub struct CrawlerConfig {
26    /// Seed URL (must be HTTP/HTTPS). Mutually exclusive with `url_list`
27    /// and `remote_url_list`.
28    #[serde(skip_serializing_if = "String::is_empty")]
29    pub url: String,
30    /// Explicit URL list (no discovery). Mutually exclusive with `url`
31    /// and `remote_url_list`. When set, the SDK posts the request as
32    /// multipart/form-data so the URLs are uploaded as a streamed file
33    /// payload rather than inlined into the JSON body.
34    #[serde(skip)]
35    pub url_list: Vec<String>,
36    /// URL of a hosted text file (one URL per line) fetched at crawl
37    /// start. Mutually exclusive with `url` and `url_list`.
38    #[serde(skip_serializing_if = "String::is_empty")]
39    pub remote_url_list: String,
40
41    /// Max pages to crawl.
42    #[serde(skip_serializing_if = "Option::is_none")]

44    /// Max link-follow depth.

231    /// Validate numeric bounds + list sizes. Ported from
232    /// `sdk/go/config_crawler.go::validateBounds`.
233    pub fn validate(&self) -> Result<(), ScrapflyError> {
234        let has_seed = !self.url.is_empty();
235        let has_list = !self.url_list.is_empty();
236        let has_remote = !self.remote_url_list.is_empty();
237        let count = (has_seed as u8) + (has_list as u8) + (has_remote as u8);
238        if count == 0 {
239            return Err(ScrapflyError::Config(
240                "provide one of url, url_list, or remote_url_list".into(),
241            ));
242        }
243        if count > 1 {
244            return Err(ScrapflyError::Config(
245                "only one of url, url_list, or remote_url_list can be set".into(),
246            ));
247        }
248        if let Some(d) = self.max_duration {
249            if !(15..=10800).contains(&d) {
250                return Err(ScrapflyError::Config(format!(
251                    "max_duration must be between 15 and 10800 seconds, got {}",
252                    d
253                )));
254            }
255        }
256        if let Some(rd) = self.rendering_delay {
257            if rd > 25000 {
258                return Err(ScrapflyError::Config(format!(
259                    "rendering_delay must be between 0 and 25000 ms, got {}",
260                    rd
261                )));
262            }
263        }
264        if let Some(delay) = self.delay {
265            if delay > 15000 {
266                return Err(ScrapflyError::Config(format!(
267                    "delay must be between 0 and 15000 ms, got {}",
268                    delay
269                )));
270            }
271        }
272        if let Some(ttl) = self.cache_ttl {
273            if ttl > 604800 {
274                return Err(ScrapflyError::Config(format!(
275                    "cache_ttl must be between 0 and 604800 seconds, got {}",
276                    ttl
277                )));
278            }
279        }
280        if self.exclude_paths.len() > 100 {
281            return Err(ScrapflyError::Config(format!(
282                "exclude_paths is limited to 100 entries, got {}",
283                self.exclude_paths.len()
284            )));
285        }
286        if self.include_only_paths.len() > 100 {
287            return Err(ScrapflyError::Config(format!(
288                "include_only_paths is limited to 100 entries, got {}",
289                self.include_only_paths.len()
290            )));
291        }
292        if !self.exclude_paths.is_empty() && !self.include_only_paths.is_empty() {
293            return Err(ScrapflyError::Config(
294                "exclude_paths and include_only_paths are mutually exclusive".into(),
295            ));
296        }
297        if self.allowed_external_domains.len() > 250 {
298            return Err(ScrapflyError::Config(format!(
299                "allowed_external_domains is limited to 250 entries, got {}",
300                self.allowed_external_domains.len()
301            )));
302        }
303        if self.allowed_internal_subdomains.len() > 250 {
304            return Err(ScrapflyError::Config(format!(
305                "allowed_internal_subdomains is limited to 250 entries, got {}",
306                self.allowed_internal_subdomains.len()
307            )));
308        }
309        if let Some(interval) = self.refresh_interval {
310            if !(REFRESH_MIN_INTERVAL..=REFRESH_MAX_INTERVAL).contains(&interval) {
311                return Err(ScrapflyError::Config(format!(
312                    "refresh_interval must be between {} and {} seconds, got {}",
313                    REFRESH_MIN_INTERVAL, REFRESH_MAX_INTERVAL, interval
314                )));
315            }
316            // A period with the feature off would silently never run.
317            if !self.refresh {
318                return Err(ScrapflyError::Config(
319                    "refresh_interval requires refresh = true".into(),
320                ));
321            }
322        }
323        Ok(())
324    }

326    /// Serialize into the JSON body the crawler endpoint expects. Used
327    /// for seed-URL crawls and remote_url_list crawls. For in-memory URL
328    /// lists the SDK switches to multipart (see `to_multipart_body`).
329    pub fn to_json_body(&self) -> Result<Vec<u8>, ScrapflyError> {
330        self.validate()?;
331        Ok(serde_json::to_vec(self)?)
332    }
333
334    /// Build a multipart/form-data body for `POST /crawl` when an in-memory
335    /// `url_list` is supplied. The `config` part carries the JSON config
336    /// (without url_list) and the `urls` part carries the URLs as
337    /// text/plain, one per line. Returns the body bytes and the matching
338    /// Content-Type header (with the boundary baked in).
339    pub fn to_multipart_body(&self) -> Result<(Vec<u8>, String), ScrapflyError> {
340        self.validate()?;
341        if self.url_list.is_empty() {
342            return Err(ScrapflyError::Config(
343                "to_multipart_body requires url_list to be set".into(),
344            ));
345        }
346
347        // Boundary derived from the SDK name + a short random tail. The
348        // value is opaque to the server; it just needs to be unique within
349        // the request and absent from the body content (URLs and JSON).
350        let boundary = format!("scrapfly-rs-{:016x}", rand_u64());
351        let config_json = serde_json::to_vec(self)?;
352        let urls_blob = self.url_list.join("\n");
353
354        let mut buf: Vec<u8> = Vec::with_capacity(config_json.len() + urls_blob.len() + 256);
355        buf.extend_from_slice(format!("--{}\r\n", boundary).as_bytes());
356        buf.extend_from_slice(
357            b"Content-Disposition: form-data; name=\"config\"; filename=\"config.json\"\r\n",
358        );
359        buf.extend_from_slice(b"Content-Type: application/json\r\n\r\n");
360        buf.extend_from_slice(&config_json);
361        buf.extend_from_slice(b"\r\n");
362        buf.extend_from_slice(format!("--{}\r\n", boundary).as_bytes());
363        buf.extend_from_slice(
364            b"Content-Disposition: form-data; name=\"urls\"; filename=\"urls.txt\"\r\n",
365        );
366        buf.extend_from_slice(b"Content-Type: text/plain\r\n\r\n");
367        buf.extend_from_slice(urls_blob.as_bytes());
368        buf.extend_from_slice(b"\r\n");
369        buf.extend_from_slice(format!("--{}--\r\n", boundary).as_bytes());
370
371        Ok((buf, format!("multipart/form-data; boundary={}", boundary)))
372    }
373}

375// Small non-crypto random for the multipart boundary. We use a fast,
376// dependency-free PRNG seeded from the system clock — the boundary just
377// needs to be unique within the request, not cryptographically secure.
378fn rand_u64() -> u64 {
379    use std::time::{SystemTime, UNIX_EPOCH};
380    let nanos = SystemTime::now()
381        .duration_since(UNIX_EPOCH)
382        .map(|d| d.as_nanos() as u64)
383        .unwrap_or(0);
384    // splitmix64 step from the source paper — adequate for boundary tags.
385    let mut z = nanos.wrapping_add(0x9E3779B97F4A7C15);
386    z = (z ^ (z >> 30)).wrapping_mul(0xBF58476D1CE4E5B9);
387    z = (z ^ (z >> 27)).wrapping_mul(0x94D049BB133111EB);
388    z ^ (z >> 31)
389}

391/// Builder for [`CrawlerConfig`].
392#[derive(Debug, Clone)]
393pub struct CrawlerConfigBuilder {
394    cfg: CrawlerConfig,
395    // `CrawlerConfig` has no `unblocker` field — it derives Serialize, so a
396    // field of that name would put `unblocker` in the POST /crawl body. The
397    // input is parked here and folded onto `cfg.asp` at build() time.
398    unblocker_input: bool,
399    // Which of the two input names the caller actually reached for. A plain
400    // `bool` cannot tell "not supplied" from "supplied false", and the
401    // decided precedence needs that distinction to let `.asp(false)` veto
402    // `.unblocker(true)` regardless of call order.
403    asp_set: bool,
404    unblocker_set: bool,
405}

578    /// Finalize the builder.
579    pub fn build(self) -> Result<CrawlerConfig, ScrapflyError> {
580        let CrawlerConfigBuilder {
581            mut cfg,
582            unblocker_input,
583            asp_set,
584            unblocker_set,
585        } = self;
586
587        // Fixed precedence, identical in every Scrapfly SDK: a supplied
588        // `asp` wins, `unblocker` decides only when `asp` was not supplied.
589        // The result lands on `cfg.asp` because that field name is the wire
590        // key (this struct derives Serialize) and the wire key does not
591        // change in this release.
592        if !asp_set && unblocker_set {
593            cfg.asp = unblocker_input;
594        }
595
596        cfg.validate()?;
597        Ok(cfg)
598    }
